cmake_minimum_required(VERSION 3.24)
project(KANConsumer LANGUAGES CXX)
option(KAN_CONSUMER_CUDA "Also exercise an installed CUDA backend on real hardware" OFF)
find_package(KAN 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE kan::kan)
if(KAN_CONSUMER_CUDA)
    target_link_libraries(consumer PRIVATE kan::cuda)
    target_compile_definitions(consumer PRIVATE KAN_CONSUMER_CUDA)
endif()
enable_testing()
add_test(NAME installed_package COMMAND consumer)
